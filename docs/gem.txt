      method: 'POST',

        generationConfig: {
            responseMimeType: "application/json",
            responseSchema: {
                type: "OBJECT",
                properties: {
                    summary: { type: "STRING" },
                    nextAction: { type: "STRING" }
                }
            }
        }
      })
    });
    const data = await response.json();
    return JSON.parse(data.candidates[0].content.parts[0].text);
  } catch (error) {
    console.error("AI生成エラー:", error);
    return { summary: "生成に失敗しました。", nextAction: "しばらく経ってから再度お試しください。" };
  }
};

// Helper: クローズ判定
const isCaseClosed = (resultStr) => {
  if (!resultStr) return false;
  const s = resultStr.toLowerCase();
  return s.includes('close') || s.includes('完了') || s.includes('クローズ');
};

// 表記揺れ吸収用の文字正規化ヘルパー
const normalizeKey = (key) => {
  if (!key) return '';
  return key.replace(/[\s\(\)（）\-_\/]/g, '').toLowerCase();
};

export default function App() {
  const [user, setUser] = useState(null);
  const [cases, setCases] = useState([]);
  const [loading, setLoading] = useState(true);
  
  // シード済みフラグ（削除時にモックデータが無限復活するのを防ぐ）
  const [hasSeeded, setHasSeeded] = useState(() => localStorage.getItem('legal_hub_seeded') === 'true');

  const [searchQuery, setSearchQuery] = useState("");
  const [sortConfig, setSortConfig] = useState({ key: 'lastUpdated', direction: 'desc' });

  };

  const handleMoveColumn = (index, direction) => {
    if (direction === 'left' && index === 0) return;
    if (direction === 'right' && index === tableColumns.length - 1) return;
    const nextIndex = direction === 'left' ? index - 1 : index + 1;
    const updated = [...tableColumns];
    const temp = updated[index];
    updated[index] = updated[nextIndex];
    updated[nextIndex] = temp;
    saveColumnSettings(updated);
  };

            if (!Array.isArray(list)) return [];
            return list.map((item, idx) => ({
              id: item.id || `safe-id-${idx}-${Date.now()}`,
              ...item
            }));
          };

          return { 
            id: doc.id, 
            ...data,
            history: sanitizeList(data.history),
            minutes: sanitizeList(data.minutes),

        setCases(fetchedCases);
        if (selectedCase) {
          const updatedSelected = fetchedCases.find(c => c.id === selectedCase.id);
          if (updatedSelected) setSelectedCase(updatedSelected);
        }
        setLoading(false);
      }
    }, (error) => {
      console.error("Firestore error:", error);
      if (cases.length === 0) setCases(initialCases);
      setLoading(false);
    });

    return () => unsubscribe();

  // --- GAS 一括インポート (表記揺れ＆自動要約付きマッピング) ---
  const handleImportFromGas = async () => {
    if (!gasUrl) {
      setSyncStatus('error');
      setSyncErrorMessage('GASウェブアプリURLを設定してください。');
      return;
    }
    setSyncStatus('syncing');

    try {
      const targetUrl = `${gasUrl}?apiKey=${encodeURIComponent(gasApiKey)}`;
      const response = await fetch(targetUrl);
      if (!response.ok) throw new Error("GASサーバーがエラーを返しました。");
      const data = await response.json();
      if (data.status === 'error') throw new Error(data.message || '認証エラーが発生しました。');

      if (Array.isArray(data)) {
        const casesRef = collection(db, 'artifacts', appId, 'public', 'data', 'legal_cases');
        
        if (!hasSeeded) {
          setHasSeeded(true);
          localStorage.setItem('legal_hub_seeded', 'true');
        }

        for (const item of data) {
          const itemMap = {};
          Object.keys(item).forEach(k => {
            itemMap[normalizeKey(k)] = item[k];
          });

          const caseName = itemMap['案件名企業名'] || itemMap['案件名'] || itemMap['企業名'] || itemMap['casename'] || itemMap['companyname'] || '';
          const naDate = itemMap['na日'] || itemMap['na'] || itemMap['nadate'] || '';
          const startDate = itemMap['対応開始日'] || itemMap['startdate'] || '';
          const priority = itemMap['優先度'] || itemMap['priority'] || '中';
          const chief = itemMap['担当chief'] || itemMap['担当'] || itemMap['chief'] || '';
          const ball = itemMap['ボール'] || itemMap['ball'] || '';
          const naLedger = itemMap['na台帳'] || itemMap['na処理台帳'] || itemMap['naledger'] || '';
          const threadUrl = itemMap['スレッドurl'] || itemMap['threadurl'] || '';
          
          const issueSummary = itemMap['問題箇所1行'] || itemMap['問題箇所'] || itemMap['問題該当箇所要約'] || itemMap['issuesummary'] || '';
          const issueDetails = itemMap['問題該当箇所'] || itemMap['問題該当箇所詳細'] || itemMap['issuedetails'] || '';
          const contractDetails = itemMap['契約書箇所'] || itemMap['契約書及び利用規約該当箇所'] || itemMap['契約書'] || itemMap['contractdetails'] || '';
          const direction = itemMap['方向性'] || itemMap['direction'] || '';
          
          const rawResult = itemMap['結果'] || itemMap['result'] || '';
          const result = (!rawResult || String(rawResult).trim() === '') ? '対応中' : String(rawResult).trim();
          
          const rawId = itemMap['id'] || itemMap['id列'] || `case-${Math.random().toString(36).substr(2, 9)}`;
          const docId = String(rawId).trim();

          const parseArrayData = (rawData) => {
            if (!rawData) return [];
            try {
              const list = typeof rawData === 'string' ? JSON.parse(rawData) : rawData;
              if (Array.isArray(list)) {
                return list.map((l, i) => ({
                  id: l.id || `gas-rec-${i}-${Date.now()}`,
                  ...l
                }));
              }
              return [];
            } 
            catch (e) { return [{ id: `gas-raw-${Date.now()}`, date: new Date().toLocaleDateString(), type: 'スプシ', content: String(rawData) }]; }
          };

          let finalIssueSummary = String(issueSummary).trim();
          const finalIssueDetails = String(issueDetails).trim();
          if (finalIssueSummary === '' && finalIssueDetails !== '') {
            finalIssueSummary = finalIssueDetails.length > 45 
              ? finalIssueDetails.substring(0, 45) + '...' 
              : finalIssueDetails;
          }

          const formatted = {
            id: docId,
            caseName: String(caseName).trim(),
            naDate: String(naDate).trim(),
            startDate: String(startDate).trim(),
            priority: String(priority).trim(),
            chief: String(chief).trim(),
            ball: String(ball).trim(),
            issueSummary: finalIssueSummary,
            naLedger: String(naLedger).trim(),
            issueDetails: finalIssueDetails,
            contractDetails: String(contractDetails).trim(),
            direction: String(direction).trim(),
            threadUrl: String(threadUrl).trim(),
            result: result,
            lastUpdated: item.lastUpdated ? Number(item.lastUpdated) : new Date().getTime(),
            history: parseArrayData(item.history || item['対応履歴']),
            minutes: parseArrayData(item.minutes || item['議事録履歴'])
          };

          if (formatted.caseName) {
            await setDoc(doc(casesRef, docId), formatted);
          }
        }
        setSyncStatus('success');
      } else {
        throw new Error("取得データが正しい配列形式ではありません。");
      }
    } catch (err) {
      console.error("GAS Import error:", err);
      setSyncStatus('error');
      setSyncErrorMessage(err.message || '読み込みに失敗しました。');
    }
  };

  // --- GAS 一括エクスポート (バックアップ) ---
  const handleExportToGas = async () => {
    if (!gasUrl) {
      setSyncStatus('error');
      setSyncErrorMessage('GASウェブアプリURLを設定してください。');
      return;

      await fetch(gasUrl, {
        method: 'POST',
        mode: 'no-cors', 
        headers: { 'Content-Type': 'text/plain' },
        body: JSON.stringify({
          apiKey: gasApiKey,
          action: 'exportAll',
          cases: cases
        })
      });
      setSyncStatus('success');
      setTimeout(() => setSyncStatus('idle'), 3000);
    } catch (err) {
      console.error("GAS Export error:", err);
      setSyncStatus('error');
      setSyncErrorMessage(err.message || 'バックアップ書き出しに失敗しました。');
    }
  };

  // --- 設定保存処理 ---
  const handleSaveSettings = (url, key) => {
    const trimmedUrl = url.trim();
    const trimmedKey = key.trim();
    setGasUrl(trimmedUrl);
    setGasApiKey(trimmedKey);
    localStorage.setItem('legal_assistant_gas_url_v3', trimmedUrl);
    localStorage.setItem('legal_assistant_gas_key_v3', trimmedKey);

    setTimeout(() => setSyncStatus('idle'), 3000);
  };

  // --- データハンドラー ---
  const handleUpdateCase = async (id, field, value) => {
    const updatedTime = new Date().getTime();
    
    // 楽観的アップデート
    setCases(prev => prev.map(c => c.id === id ? { ...c, [field]: value, lastUpdated: updatedTime } : c));
    if (selectedCase && selectedCase.id === id) {
      setSelectedCase(prev => ({ ...prev, [field]: value, lastUpdated: updatedTime }));
    }

    if (user) {
      const caseRef = doc(db, 'artifacts', appId, 'public', 'data', 'legal_cases', id);
      try {
        await updateDoc(caseRef, { [field]: value, lastUpdated: updatedTime });
      } catch (err) { console.error("Firestore update error:", err); }
    }
  };

  const handleAddListRecord = async (caseId, fieldName, record) => {
    const targetCase = cases.find(c => c.id === caseId);
    if (!targetCase) return;
    const currentList = targetCase[fieldName] || [];
    const newList = [...currentList, { id: `rec-${Date.now()}`, ...record }];
    await handleUpdateCase(caseId, fieldName, newList);
  };

    if (!window.confirm('この記録を削除してもよろしいですか？')) return;
    const targetCase = cases.find(c => c.id === caseId);
    if (!targetCase) return;
    const newList = (targetCase[fieldName] || []).filter(r => r.id !== recordId);
    await handleUpdateCase(caseId, fieldName, newList);
  };

  const handleCreateNewCase = async () => {
    if (!user) return;
    const newId = `case-${Date.now()}`;
    const newCase = {
      id: newId,
      caseName: "新規案件",
      naDate: "",
      startDate: new Date().toISOString().split('T')[0],
      priority: "中",
      chief: "",
      ball: "",
      issueSummary: "",
      naLedger: "",
      issueDetails: "",
      contractDetails: "",
      direction: "",
      threadUrl: "",
      result: "対応中",
      lastUpdated: new Date().getTime(),
      history: [],
      minutes: []
    };
    
    if (!hasSeeded) {
      setHasSeeded(true);
      localStorage.setItem('legal_hub_seeded', 'true');
    }

    const caseRef = doc(db, 'artifacts', appId, 'public', 'data', 'legal_cases', newId);
    await setDoc(caseRef, newCase);
    setCases(prev => [newCase, ...prev]);
    setSelectedCase(newCase);
    setActiveTab('detail');
  };

  // 案件の完全削除
  const handleDeleteCase = async (id) => {
    if (!window.confirm('この案件を完全に削除してもよろしいですか？\n※この操作は取り消せません。')) return;
    
    if (!hasSeeded) {
      setHasSeeded(true);
      localStorage.setItem('legal_hub_seeded', 'true');
    }

    // 楽観的アップデート
    setCases(prev => prev.filter(c => c.id !== id));
    if (selectedCase && selectedCase.id === id) {
      setSelectedCase(null);
      setActiveTab('activeList');
    }

    if (user) {
      const caseRef = doc(db, 'artifacts', appId, 'public', 'data', 'legal_cases', id);
      try {
        await deleteDoc(caseRef);
        console.log("Deleted successfully.");
      } catch (err) {
        console.error("Firestore delete error:", err);
      }
    }
  };

  const handleGenerateAI = async () => {
    if (!selectedCase) return;
    setAiLoading(true);
    setAiResult(null);

    const historyStr = (selectedCase.history || []).map(h => `${h.date} [${h.type}] ${h.content}`).join('\n');
    const minutesStr = (selectedCase.minutes || []).map(m => `${m.date} [${m.type}] ${m.content}`).join('\n');

    const promptText = `
      案件名: ${selectedCase.caseName}
      現在のボール: ${selectedCase.ball}
      結果・ステータス: ${selectedCase.result}
      問題該当箇所: ${selectedCase.issueDetails}
      契約書該当箇所: ${selectedCase.contractDetails}
      方向性: ${selectedCase.direction}
      
      【対応履歴】
      ${historyStr}
      
      【議事録履歴】
      ${minutesStr}
    `;

    const result = await generateAIResponse(promptText);
    setAiResult(result);
    setAiLoading(false);
  };

  // --- Computed Data ---
  const isStale = (timestamp) => {
    const twoDaysMs = 48 * 60 * 60 * 1000;
    return (new Date().getTime() - timestamp) > twoDaysMs;
  };

  // 表示対象の絞り込み
  const displayCases = useMemo(() => {
    let result = [...cases];
    
    // タブによるフィルタリング (Closeの表示/非表示)
    if (activeTab === 'activeList') {
      result = result.filter(c => !isCaseClosed(c.result));
    }

    // 2日以上未更新フィルターがONの場合の追加絞り込み
    if (staleFilterActive) {
      result = result.filter(c => !isCaseClosed(c.result) && isStale(c.lastUpdated));
    }

    // 検索フィルタ
    if (searchQuery) {
      const lowerQuery = searchQuery.toLowerCase();
      result = result.filter(c => 
        (c.caseName && c.caseName.toLowerCase().includes(lowerQuery)) ||
        (c.chief && c.chief.toLowerCase().includes(lowerQuery)) ||
        (c.issueSummary && c.issueSummary.toLowerCase().includes(lowerQuery))
      );
    }

    // ソート
    result.sort((a, b) => {
      let valA = a[sortConfig.key];
      let valB = b[sortConfig.key];
      if (valA === undefined) valA = '';
      if (valB === undefined) valB = '';
      if (valA < valB) return sortConfig.direction === 'asc' ? -1 : 1;
      if (valA > valB) return sortConfig.direction === 'asc' ? 1 : -1;
      return 0;
    });

    return result;
  }, [cases, activeTab, searchQuery, sortConfig, staleFilterActive]);

  const staleCasesCount = cases.filter(c => !isCaseClosed(c.result) && isStale(c.lastUpdated)).length;

  const requestSort = (key) => {
    let direction = 'asc';
    if (sortConfig.key === key && sortConfig.direction === 'asc') direction = 'desc';
    setSortConfig({ key, direction });
  };

  const SortIcon = ({ columnKey }) => {
    if (sortConfig.key !== columnKey) return <ChevronDown className="w-3.5 h-3.5 text-gray-400 opacity-0 group-hover:opacity-50" />;
    return sortConfig.direction === 'asc' ? 
      <ChevronUp className="w-3.5 h-3.5 text-blue-600" /> : 
      <ChevronDown className="w-3.5 h-3.5 text-blue-600" />;
  };

  // リストUI用のコンポーネント
  const RecordListForm = ({ caseId, fieldName, title, icon: Icon, records, sortDesc, setSortDesc }) => {
    const [date, setDate] = useState(new Date().toISOString().split('T')[0]);
    const [type, setType] = useState('');
    const [content, setContent] = useState('');
    const listId = `${caseId}-${fieldName}-types`;

    const sortedRecords = useMemo(() => {
      return [...(records || [])].sort((a, b) => {
        if (a.date < b.date) return sortDesc ? 1 : -1;
        if (a.date > b.date) return sortDesc ? -1 : 1;
        return 0;
      });
    }, [records, sortDesc]);

    const handleSubmit = (e) => {
      e.preventDefault();
      if (!content) return;
      handleAddListRecord(caseId, fieldName, { date, type, content });
      setContent('');
      setType('');
    };

    return (
      <div className="bg-white border border-gray-200 rounded-xl shadow-sm overflow-hidden flex flex-col h-[500px]">
        <div className="p-3 border-b border-gray-100 bg-gray-50 flex items-center justify-between">
          <h3 className="font-bold text-gray-800 flex items-center gap-2">
            <Icon className="w-4 h-4 text-blue-600" /> {title} <span className="text-xs text-gray-500 font-normal ml-1">({records?.length || 0})</span>
          </h3>
          <button 
            type="button"
            onClick={() => setSortDesc(!sortDesc)} 
            className="text-xs flex items-center gap-1 text-gray-500 hover:text-gray-800 transition-colors"
          >
            {sortDesc ? <ChevronDown className="w-4 h-4"/> : <ChevronUp className="w-4 h-4"/>} 日付順
          </button>
        </div>
        
        {/* リスト表示領域 */}
        <div className="flex-1 overflow-y-auto p-4 space-y-4 bg-gray-50/50">
          {sortedRecords.length === 0 ? (
            <p className="text-sm text-gray-400 text-center py-8">記録はありません</p>
          ) : (
            sortedRecords.map((rec) => (
              <div key={rec.id} className="bg-white p-3 rounded-lg border border-gray-200 shadow-sm relative group">
                <div className="flex items-start justify-between mb-2">
                  <div className="flex items-center gap-2">
                    <span className="text-xs font-semibold text-gray-500 flex items-center gap-1"><Calendar className="w-3 h-3"/> {rec.date}</span>
                    {rec.type && <span className="text-[10px] bg-blue-50 text-blue-700 px-2 py-0.5 rounded font-bold border border-blue-100">{rec.type}</span>}
                  </div>
                  <button 
                    type="button"
                    onClick={() => handleDeleteListRecord(caseId, fieldName, rec.id)}
                    className="text-gray-300 hover:text-red-500 opacity-0 group-hover:opacity-100 transition-opacity p-1"
                    title="削除"
                  >
                    <Trash2 className="w-3.5 h-3.5" />
                  </button>
                </div>
                <p className="text-sm text-gray-800 whitespace-pre-wrap leading-relaxed">{rec.content}</p>
              </div>
            ))
          )}
        </div>

        {/* 入力フォーム */}
        <div className="p-3 border-t border-gray-200 bg-white">
          <form onSubmit={handleSubmit} className="flex flex-col gap-2">
            <div className="flex gap-2">
              <input 
                type="date" required value={date} onChange={e => setDate(e.target.value)}
                className="text-xs p-2 border border-gray-300 rounded focus:ring-1 focus:ring-blue-500 w-32"
              />
              <input 
                type="text" list={listId} placeholder="種別 (入力or選択)" value={type} onChange={e => setType(e.target.value)}
                className="text-xs p-2 border border-gray-300 rounded focus:ring-1 focus:ring-blue-500 flex-1"
              />
              <datalist id={listId}>
                <option value="メール受信" />
                <option value="メール送信" />
                <option value="社内MTG" />
                <option value="社外MTG" />
                <option value="電話" />
                <option value="チャット" />
                <option value="ドラフト作成" />
                <option value="レビュー" />
              </datalist>
            </div>
            <div className="flex gap-2">
              <textarea 
                required placeholder="内容を入力..." value={content} onChange={e => setContent(e.target.value)}
                className="text-sm p-2 border border-gray-300 rounded focus:ring-1 focus:ring-blue-500 flex-1 h-10 min-h-[40px] resize-y"
              />
              <button type="submit" className="bg-blue-600 hover:bg-blue-700 text-white px-3 py-2 rounded text-xs font-bold transition-colors flex items-center justify-center shrink-0">
                <Plus className="w-4 h-4" />
              </button>
            </div>
          </form>
        </div>
      </div>
    );
  };

  return (
    <div className="min-h-screen bg-gray-50 flex flex-col font-sans text-gray-800">
      {/* --- Header --- */}
      <header className="bg-white border-b border-gray-200 px-6 py-4 flex items-center justify-between sticky top-0 z-20 shadow-sm">
        <div className="flex items-center gap-3">
          <div className="bg-blue-600 p-2 rounded-lg shadow-inner">
            <Bot className="w-6 h-6 text-white" />
          </div>
          <div>
            <h1 className="text-xl font-extrabold text-gray-900 tracking-tight">Legal Assist Hub</h1>
            <p className="text-[10px] text-gray-500 font-semibold uppercase tracking-wider">法務部 案件進捗管理システム</p>
          </div>
        </div>
        
        <div className="flex items-center gap-4">
          <button 
            onClick={() => setShowGasSettings(!showGasSettings)}
            className={`flex items-center gap-1.5 px-3 py-1.5 rounded-md text-xs font-bold transition-all border ${
              gasUrl ? 'bg-blue-50 text-blue-700 border-blue-200 hover:bg-blue-100' : 'bg-amber-50 text-amber-700 border-amber-200 hover:bg-amber-100'
            }`}
          >
            <Settings className="w-4 h-4" /> {gasUrl ? 'バックアップ設定済' : 'バックアップ未設定'}
          </button>

          {/* 2日以上未更新アラート 兼 フィルターボタン */}
          {staleCasesCount > 0 && (
            <button 
              onClick={() => setStaleFilterActive(!staleFilterActive)}
              className={`flex items-center gap-1.5 px-3.5 py-1.5 rounded-full border text-xs font-bold transition-all ${
                staleFilterActive 
                  ? 'bg-amber-500 text-white border-amber-600 shadow-md scale-105' 
                  : 'bg-amber-50 text-amber-700 border-amber-200 shadow-sm animate-pulse hover:bg-amber-100'
              }`}
              title="クリックして未更新案件のみをフィルタリング"
            >
              <AlertTriangle className="w-4 h-4" />
              <span>{staleFilterActive ? '未更新のみ表示中' : `2日以上未更新: ${staleCasesCount}件`}</span>
              {staleFilterActive && <X className="w-3 h-3 ml-1 bg-amber-600 rounded-full" />}
            </button>
          )}
          
          <nav className="flex bg-gray-100 p-1 rounded-lg">
            <button 
              onClick={() => { setActiveTab('activeList'); setSelectedCase(null); setStaleFilterActive(false); }}
              className={`px-3 py-1.5 text-xs font-bold rounded-md transition-colors ${activeTab === 'activeList' || (activeTab === 'detail' && !isCaseClosed(selectedCase?.result)) ? 'bg-white shadow-sm text-blue-700' : 'text-gray-500 hover:text-gray-800'}`}
            >
              案件管理 (対応中)
            </button>
            <button 
              onClick={() => { setActiveTab('allList'); setSelectedCase(null); setStaleFilterActive(false); }}
              className={`px-3 py-1.5 text-xs font-bold rounded-md transition-colors ${activeTab === 'allList' || (activeTab === 'detail' && isCaseClosed(selectedCase?.result)) ? 'bg-white shadow-sm text-blue-700' : 'text-gray-500 hover:text-gray-800'}`}
            >
              全案件一覧
            </button>
            <button 
              onClick={() => { setActiveTab('sheet'); setStaleFilterActive(false); }}
              className={`px-3 py-1.5 text-xs font-bold rounded-md transition-colors flex items-center gap-1 ${activeTab === 'sheet' ? 'bg-white shadow-sm text-blue-700' : 'text-gray-500 hover:text-gray-800'}`}
            >
              元データ書き出し
            </button>
          </nav>
        </div>
      </header>

      {/* --- Backup Settings Panel --- */}
      {showGasSettings && (
        <div className="bg-white border-b border-gray-200 p-6 shadow-inner animate-in slide-in-from-top duration-200 z-10">
          <div className="max-w-4xl mx-auto space-y-4">
            <div className="flex items-center justify-between border-b border-gray-100 pb-2">
              <h3 className="text-base font-bold text-gray-900 flex items-center gap-2">
                <Database className="w-5 h-5 text-blue-600" />
                スプレッドシート・バックアップ設定
              </h3>
              <span className="text-xs text-gray-500">※普段の入力は自動でシステム（Firestore）に安全に保存されます。</span>
            </div>

            <div className="grid grid-cols-2 gap-4">
              <div>
                <label className="block text-xs font-bold text-gray-600 mb-1">① GAS ウェブアプリURL</label>
                <input 
                  type="text" id="settings-gas-url" defaultValue={gasUrl} placeholder="https://script.google.com/macros/s/.../exec"
                  className="w-full p-2 text-sm border border-gray-300 rounded bg-gray-50 focus:bg-white focus:ring-2 focus:ring-blue-500 font-mono"
                />
              </div>
              <div>
                <label className="block text-xs font-bold text-gray-600 mb-1 flex items-center gap-1">
                  <Key className="w-3.5 h-3.5 text-amber-500" /> ② APIキー (パスワード)
                </label>
                <input 
                  type="password" id="settings-gas-key" defaultValue={gasApiKey} placeholder="例: legal_secret_pass"
                  className="w-full p-2 text-sm border border-gray-300 rounded bg-gray-50 focus:bg-white focus:ring-2 focus:ring-blue-500 font-mono"
                />
              </div>
            </div>

            <div className="flex items-center gap-3 mt-4">
              <button 
                onClick={() => handleSaveSettings(
                  document.getElementById('settings-gas-url').value, 
                  document.getElementById('settings-gas-key').value
                )}
                className="bg-gray-800 text-white px-4 py-2 rounded text-sm font-bold hover:bg-gray-700"
              >
                設定を保存
              </button>
              {syncStatus === 'syncing' && <span className="text-sm text-blue-600 flex items-center gap-1"><RefreshCw className="w-4 h-4 animate-spin" /> 通信中...</span>}
              {syncStatus === 'success' && <span className="text-sm text-green-600 flex items-center gap-1"><CheckCircle className="w-4 h-4" /> 完了</span>}
              {syncStatus === 'error' && <span className="text-sm text-red-600 flex items-center gap-1"><ShieldAlert className="w-4 h-4" /> {syncErrorMessage}</span>}
              
              <div className="ml-auto flex gap-2">
                <button onClick={handleImportFromGas} className="bg-white border border-gray-300 text-gray-700 px-3 py-2 rounded text-sm font-bold hover:bg-gray-50 flex items-center gap-1">
                  <ArrowDownToLine className="w-4 h-4" /> スプシから読込
                </button>
                <button onClick={handleExportToGas} className="bg-blue-50 border border-blue-200 text-blue-700 px-3 py-2 rounded text-sm font-bold hover:bg-blue-100 flex items-center gap-1">
                  <ArrowUpFromLine className="w-4 h-4" /> スプシへ書出
                </button>
              </div>
            </div>
          </div>
        </div>
      )}

      {/* --- Main Content --- */}
      <main className="flex-1 overflow-hidden flex flex-col relative">
        {loading ? (
          <div className="flex-1 flex items-center justify-center">
            <RefreshCw className="w-8 h-8 text-blue-600 animate-spin" />
          </div>
        ) : (
          <>
            {/* List View */}
            {(activeTab === 'activeList' || activeTab === 'allList') && (
              <div className="flex-1 flex flex-col h-full overflow-hidden p-6 gap-4">
                <div className="flex items-center justify-between">
                  <div className="relative w-96">
                    <Search className="w-5 h-5 text-gray-400 absolute left-3 top-1/2 -translate-y-1/2" />
                    <input 
                      type="text" placeholder="案件名、担当者、問題箇所で検索..." 
                      value={searchQuery} onChange={e => setSearchQuery(e.target.value)}
                      className="w-full pl-10 pr-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 outline-none text-sm"
                    />
                  </div>
                  <div className="flex gap-2">
                    <div className="relative">
                      <button onClick={() => setShowColumnSettings(!showColumnSettings)} className="bg-white border border-gray-300 text-gray-700 px-3 py-2 rounded-lg text-sm font-bold hover:bg-gray-50 flex items-center gap-2 shadow-sm">
                        <Settings className="w-4 h-4" /> 表示列設定
                      </button>
                      {showColumnSettings && (
                        <div className="absolute right-0 top-full mt-1 w-64 bg-white border border-gray-200 shadow-xl rounded-lg z-30 p-2">
                          <div className="text-xs font-bold text-gray-500 mb-2 px-2 border-b pb-1">列の表示/非表示</div>
                          <div className="max-h-64 overflow-y-auto">
                            {tableColumns.map((col, idx) => (
                              <div key={col.key} className="flex items-center justify-between px-2 py-1.5 hover:bg-gray-50 rounded group">
                                <label className="flex items-center gap-2 cursor-pointer text-sm flex-1">
                                  <input type="checkbox" checked={col.visible} onChange={() => handleToggleColumnVisible(col.key)} className="rounded text-blue-600" />
                                  {col.label}
                                </label>
                                <div className="flex gap-1 opacity-0 group-hover:opacity-100 transition-opacity">
                                  <button onClick={() => handleMoveColumn(idx, 'left')} disabled={idx === 0} className="p-0.5 text-gray-400 hover:text-gray-800 disabled:opacity-30"><ArrowLeft className="w-3 h-3" /></button>
                                  <button onClick={() => handleMoveColumn(idx, 'right')} disabled={idx === tableColumns.length - 1} className="p-0.5 text-gray-400 hover:text-gray-800 disabled:opacity-30"><ArrowRight className="w-3 h-3" /></button>
                                </div>
                              </div>
                            ))}
                          </div>
                        </div>
                      )}
                    </div>
                    <button onClick={handleCreateNewCase} className="bg-blue-600 hover:bg-blue-700 text-white px-4 py-2 rounded-lg text-sm font-bold flex items-center gap-2 shadow-sm transition-colors">
                      <Plus className="w-4 h-4" /> 新規案件
                    </button>
                  </div>
                </div>

                <div className="bg-white rounded-xl shadow-sm border border-gray-200 flex-1 overflow-hidden flex flex-col">
                  <div className="overflow-x-auto flex-1">
                    <table className="w-full text-left border-collapse min-w-max">
                      <thead className="bg-gray-50 text-gray-600 text-xs uppercase sticky top-0 z-10 shadow-sm">
                        <tr>
                          {tableColumns.filter(c => c.visible).map(col => (
                            <th key={col.key} className={`p-3 font-bold border-b border-gray-200 cursor-pointer hover:bg-gray-100 group ${col.width}`} onClick={() => requestSort(col.key)}>
                              <div className="flex items-center gap-1">
                                {col.label} <SortIcon columnKey={col.key} />
                              </div>
                            </th>
                          ))}
                        </tr>
                      </thead>
                      <tbody className="text-sm divide-y divide-gray-100">
                        {displayCases.map(item => (
                          <tr key={item.id} onClick={() => { setSelectedCase(item); setActiveTab('detail'); }} className="hover:bg-blue-50/50 cursor-pointer transition-colors group">
                            {tableColumns.filter(c => c.visible).map(col => {
                              let val = item[col.key];
                              if (col.key === 'lastUpdated' && val) val = new Date(val).toLocaleDateString();
                              
                              if (col.key === 'result') {
                                const closed = isCaseClosed(val);
                                return (
                                  <td key={col.key} className="p-3">
                                    <span className={`inline-flex items-center px-2 py-0.5 rounded text-xs font-bold ${closed ? 'bg-gray-100 text-gray-600' : 'bg-blue-100 text-blue-800'}`}>
                                      {val}
                                    </span>
                                  </td>
                                );
                              }
                              
                              if (col.key === 'priority') {
                                const colors = { '高': 'text-red-600 bg-red-50 border-red-100', '中': 'text-amber-600 bg-amber-50 border-amber-100', '低': 'text-emerald-600 bg-emerald-50 border-emerald-100' };
                                const colorClass = colors[val] || 'text-gray-600 bg-gray-50 border-gray-200';
                                return (
                                  <td key={col.key} className="p-3">
                                    <span className={`px-2 py-0.5 rounded text-xs font-bold border ${colorClass}`}>{val}</span>
                                  </td>
                                );
                              }

                              if (col.key === 'caseName') {
                                return (
                                  <td key={col.key} className="p-3 font-bold text-gray-900 group-hover:text-blue-700">
                                    {val}
                                  </td>
                                );
                              }

                              return <td key={col.key} className="p-3 text-gray-700 truncate max-w-[200px]" title={val}>{val}</td>;
                            })}
                          </tr>
                        ))}
                        {displayCases.length === 0 && (
                          <tr>
                            <td colSpan={tableColumns.filter(c=>c.visible).length} className="p-8 text-center text-gray-400">
                              該当する案件がありません
                            </td>
                          </tr>
                        )}
                      </tbody>
                    </table>
                  </div>
                </div>
              </div>
            )}

            {/* Detail View */}
            {activeTab === 'detail' && selectedCase && (
              <div className="flex-1 flex flex-col lg:flex-row overflow-hidden bg-gray-50">
                {/* Left Panel: Form */}
                <div className="flex-[3] p-6 overflow-y-auto border-r border-gray-200 bg-white">
                  <div className="flex items-center justify-between mb-6 pb-4 border-b border-gray-100">
                    <div className="flex items-center gap-3">
                      <button onClick={() => {setActiveTab(isCaseClosed(selectedCase.result) ? 'allList' : 'activeList'); setSelectedCase(null);}} className="p-2 hover:bg-gray-100 rounded-full transition-colors text-gray-500">
                        <ArrowLeft className="w-5 h-5" />
                      </button>
                      <h2 className="text-2xl font-extrabold text-gray-900 flex items-center gap-2">
                        <FileText className="w-6 h-6 text-blue-600" />
                        <input type="text" value={selectedCase.caseName} onChange={e => handleUpdateCase(selectedCase.id, 'caseName', e.target.value)} className="border-none focus:ring-0 p-0 font-extrabold text-2xl w-[400px] hover:bg-gray-50 rounded" placeholder="案件名を入力" />
                      </h2>
                    </div>
                    <div className="flex items-center gap-3">
                       <button onClick={() => handleDeleteCase(selectedCase.id)} className="text-red-500 hover:text-white hover:bg-red-500 px-3 py-1.5 rounded text-sm font-bold transition-colors border border-red-200">削除</button>
                    </div>
                  </div>

                  <div className="grid grid-cols-2 md:grid-cols-4 gap-4 mb-6">
                    <div className="bg-gray-50 p-3 rounded-lg border border-gray-100">
                      <label className="text-xs font-bold text-gray-500 flex items-center gap-1 mb-1"><Calendar className="w-3.5 h-3.5"/> NA日</label>
                      <input type="date" value={selectedCase.naDate} onChange={e => handleUpdateCase(selectedCase.id, 'naDate', e.target.value)} className="w-full bg-transparent border-none p-0 text-sm font-bold focus:ring-0" />
                    </div>
                    <div className="bg-gray-50 p-3 rounded-lg border border-gray-100">
                      <label className="text-xs font-bold text-gray-500 flex items-center gap-1 mb-1"><AlertTriangle className="w-3.5 h-3.5"/> 優先度</label>
                      <select value={selectedCase.priority} onChange={e => handleUpdateCase(selectedCase.id, 'priority', e.target.value)} className="w-full bg-transparent border-none p-0 text-sm font-bold focus:ring-0 cursor-pointer">
                        <option>高</option><option>中</option><option>低</option>
                      </select>
                    </div>
                    <div className="bg-gray-50 p-3 rounded-lg border border-gray-100">
                      <label className="text-xs font-bold text-gray-500 flex items-center gap-1 mb-1"><User className="w-3.5 h-3.5"/> 担当</label>
                      <input type="text" value={selectedCase.chief} onChange={e => handleUpdateCase(selectedCase.id, 'chief', e.target.value)} className="w-full bg-transparent border-none p-0 text-sm font-bold focus:ring-0" />
                    </div>
                    <div className="bg-gray-50 p-3 rounded-lg border border-gray-100">
                      <label className="text-xs font-bold text-gray-500 flex items-center gap-1 mb-1"><Briefcase className="w-3.5 h-3.5"/> ボール</label>
                      <input type="text" value={selectedCase.ball} onChange={e => handleUpdateCase(selectedCase.id, 'ball', e.target.value)} className="w-full bg-transparent border-none p-0 text-sm font-bold focus:ring-0" />
                    </div>
                  </div>

                  <div className="space-y-5">
                    <div>
                      <label className="text-sm font-bold text-gray-700 flex items-center gap-2 mb-2">
                        <Edit2 className="w-4 h-4 text-blue-500"/> 問題箇所(1行要約)
                      </label>
                      <input type="text" value={selectedCase.issueSummary} onChange={e => handleUpdateCase(selectedCase.id, 'issueSummary', e.target.value)} className="w-full p-3 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 shadow-sm text-sm" placeholder="問題の核心を1行で" />
                    </div>
                    <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                       <div>
                        <label className="text-sm font-bold text-gray-700 mb-2 block">結果・ステータス</label>
                        <input type="text" value={selectedCase.result} onChange={e => handleUpdateCase(selectedCase.id, 'result', e.target.value)} className="w-full p-2.5 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 shadow-sm text-sm font-bold bg-blue-50" />
                      </div>
                      <div>
                        <label className="text-sm font-bold text-gray-700 mb-2 block">NA処理台帳番号</label>
                        <input type="text" value={selectedCase.naLedger} onChange={e => handleUpdateCase(selectedCase.id, 'naLedger', e.target.value)} className="w-full p-2.5 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 shadow-sm text-sm" />
                      </div>
                    </div>
                    
                    <div className="grid grid-cols-1 md:grid-cols-2 gap-5">
                      <div>
                        <label className="text-sm font-bold text-gray-700 flex items-center gap-2 mb-2"><FileText className="w-4 h-4 text-amber-500"/> 契約書該当箇所</label>
                        <textarea value={selectedCase.contractDetails} onChange={e => handleUpdateCase(selectedCase.id, 'contractDetails', e.target.value)} className="w-full p-3 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 shadow-sm text-sm h-32 resize-y" placeholder="第〇条 等" />
                      </div>
                      <div>
                        <label className="text-sm font-bold text-gray-700 flex items-center gap-2 mb-2"><AlertTriangle className="w-4 h-4 text-red-500"/> 問題該当箇所詳細</label>
                        <textarea value={selectedCase.issueDetails} onChange={e => handleUpdateCase(selectedCase.id, 'issueDetails', e.target.value)} className="w-full p-3 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 shadow-sm text-sm h-32 resize-y" placeholder="詳細な問題点" />
                      </div>
                    </div>
                    
                    <div>
                      <label className="text-sm font-bold text-gray-700 flex items-center gap-2 mb-2">方向性・対応方針</label>
                      <textarea value={selectedCase.direction} onChange={e => handleUpdateCase(selectedCase.id, 'direction', e.target.value)} className="w-full p-3 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 shadow-sm text-sm h-24 resize-y" placeholder="どのように交渉・修正するか" />
                    </div>

                    <div>
                      <label className="text-sm font-bold text-gray-700 flex items-center gap-2 mb-2"><LinkIcon className="w-4 h-4 text-gray-400"/> スレッドURL</label>
                      <div className="flex gap-2">
                        <input type="url" value={selectedCase.threadUrl} onChange={e => handleUpdateCase(selectedCase.id, 'threadUrl', e.target.value)} className="flex-1 p-2.5 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 shadow-sm text-sm" placeholder="https://" />
                        {selectedCase.threadUrl && (
                          <a href={selectedCase.threadUrl} target="_blank" rel="noopener noreferrer" className="bg-gray-100 hover:bg-gray-200 text-gray-700 px-4 py-2.5 rounded-lg text-sm font-bold transition-colors flex items-center justify-center border border-gray-300">
                            <ExternalLink className="w-4 h-4" />
                          </a>
                        )}
                      </div>
                    </div>
                  </div>

                  {/* AI Summary Section */}
                  <div className="mt-8 bg-gradient-to-r from-indigo-50 to-purple-50 p-5 rounded-xl border border-indigo-100 shadow-sm">
                    <div className="flex items-center justify-between mb-4">
                      <h3 className="font-extrabold text-indigo-900 flex items-center gap-2">
                        <Sparkles className="w-5 h-5 text-indigo-500" />
                        AI アシスタント
                      </h3>
                      <button onClick={handleGenerateAI} disabled={aiLoading} className="bg-indigo-600 hover:bg-indigo-700 disabled:bg-indigo-300 text-white px-4 py-2 rounded-lg text-sm font-bold shadow transition-colors flex items-center gap-2">
                        {aiLoading ? <RefreshCw className="w-4 h-4 animate-spin" /> : <Bot className="w-4 h-4" />}
                        {aiLoading ? '推論中...' : '現状を要約・次を提案'}
                      </button>
                    </div>
                    {aiResult && (
                      <div className="space-y-4 bg-white p-4 rounded-lg shadow-sm border border-indigo-50">
                        <div>
                          <span className="text-xs font-bold text-indigo-500 uppercase tracking-wider block mb-1">これまでの経緯要約</span>
                          <p className="text-sm text-gray-800 whitespace-pre-wrap leading-relaxed">{aiResult.summary}</p>
                        </div>
                        <div className="border-t border-indigo-50 pt-4">
                          <span className="text-xs font-bold text-purple-500 uppercase tracking-wider block mb-1">推奨される次のアクション</span>
                          <p className="text-sm font-bold text-gray-900 whitespace-pre-wrap leading-relaxed">{aiResult.nextAction}</p>
                        </div>
                      </div>
                    )}
                  </div>
                </div>

                {/* Right Panel: Lists */}
                <div className="flex-[2] flex flex-col p-6 gap-6 overflow-y-auto bg-gray-100 border-l border-gray-200">
                  <RecordListForm 
                    caseId={selectedCase.id} fieldName="history" title="対応履歴" icon={Clock} 
                    records={selectedCase.history} sortDesc={historySortDesc} setSortDesc={setHistorySortDesc} 
                  />
                  <RecordListForm 
                    caseId={selectedCase.id} fieldName="minutes" title="議事録履歴" icon={FileText} 
                    records={selectedCase.minutes} sortDesc={minutesSortDesc} setSortDesc={setMinutesSortDesc} 
                  />
                </div>
              </div>
            )}

            {/* Sheet / Export View */}
            {activeTab === 'sheet' && (
              <div className="flex-1 p-8 overflow-y-auto bg-white">
                <div className="max-w-4xl mx-auto space-y-6">
                  <h2 className="text-2xl font-extrabold text-gray-900 flex items-center gap-2 border-b pb-4">
                    <Database className="w-6 h-6 text-blue-600" />
                    データエクスポート / 連携
                  </h2>
                  <div className="bg-blue-50 p-6 rounded-xl border border-blue-100 space-y-4">
                    <h3 className="font-bold text-blue-900">Google スプレッドシート連携 (GAS)</h3>
                    <p className="text-sm text-blue-800">
                      右上の「設定」からGASのURLとAPIキーを設定することで、ワンクリックで全件の一括バックアップと読み込みが可能です。<br/>
                      定期的なバックアップや、一括でのデータ修正（スプシで修正して読み込み）にご活用ください。
                    </p>
                    <div className="flex gap-4 pt-2">
                       <button onClick={handleExportToGas} className="bg-blue-600 hover:bg-blue-700 text-white px-5 py-2.5 rounded-lg text-sm font-bold shadow flex items-center gap-2">
                        <ArrowUpFromLine className="w-4 h-4" /> スプレッドシートへ全件バックアップ
                      </button>
                      <button onClick={handleImportFromGas} className="bg-white border border-gray-300 text-gray-700 hover:bg-gray-50 px-5 py-2.5 rounded-lg text-sm font-bold shadow-sm flex items-center gap-2">
                        <ArrowDownToLine className="w-4 h-4" /> スプレッドシートから読み込み
                      </button>
                    </div>
                    {syncStatus === 'syncing' && <p className="text-sm text-blue-600 flex items-center gap-1"><RefreshCw className="w-4 h-4 animate-spin" /> 連携処理を実行中...</p>}
                    {syncStatus === 'success' && <p className="text-sm text-green-600 flex items-center gap-1"><CheckCircle className="w-4 h-4" /> 処理が完了しました。</p>}
                    {syncStatus === 'error' && <p className="text-sm text-red-600 flex items-center gap-1"><ShieldAlert className="w-4 h-4" /> {syncErrorMessage}</p>}
                  </div>

                  <div className="bg-gray-50 p-6 rounded-xl border border-gray-200">
                    <h3 className="font-bold text-gray-900 mb-2">開発者用：JSONエクスポート</h3>
                    <p className="text-sm text-gray-600 mb-4">現在システムに保存されている全件データをJSON形式でクリップボードにコピーします。</p>
                    <button 
                      onClick={() => {
                        const jsonStr = JSON.stringify(cases, null, 2);
                        // navigator.clipboard is sometimes restricted in iframes, fallback to prompt if needed
                        try {
                          navigator.clipboard.writeText(jsonStr);
                          alert('クリップボードにコピーしました');
                        } catch (e) {
                          const textArea = document.createElement("textarea");
                          textArea.value = jsonStr;
                          document.body.appendChild(textArea);
                          textArea.select();
                          document.execCommand("copy");
                          textArea.remove();
                          alert('クリップボードにコピーしました');
                        }
                      }}
                      className="bg-white border border-gray-300 text-gray-700 hover:bg-gray-50 px-4 py-2 rounded-lg text-sm font-bold shadow-sm flex items-center gap-2"
                    >
                      <FileText className="w-4 h-4" /> JSONをコピー
                    </button>
                  </div>
                </div>
              </div>
            )}
          </>
        )}
      </main>
    </div>
  );
}
